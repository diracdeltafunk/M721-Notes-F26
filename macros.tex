\newcommand{\startlecture}[1][]{%
  \par\medskip\penalty-200\needspace{3\baselineskip}\noindent
  \refstepcounter{lecture}%
  \label{lecture:\thelecture}%
  \addcontentsline{lec}{lecture}{\protect\numberline{Lecture~\thelecture}#1}%
  \makebox[\linewidth]{\hrulefill\quad
    \textnormal{\footnotesize Approx. Start of Lecture \thelecture
      \if\relax\detokenize{#1}\relax\else, #1\fi}%
    \quad\hrulefill}%
  \par\medskip
}
